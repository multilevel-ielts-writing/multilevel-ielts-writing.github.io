async function callAnthropic(env, input, images, quick, maxTokens) {
if (!env.ANTHROPIC_API_KEY) return { error: "ANTHROPIC_API_KEY kaliti qöyilmagan" };
const model = quick ? cfg(env, "ANTHROPIC_QUICK_MODEL") : cfg(env, "ANTHROPIC_MODEL");

content.push({ type: "text", text: input });
let limit = Math.max(maxTokens || 0, 16000), effort = true;
const mk = () => JSON.stringify(Object.assign({ model, max_tokens: limit, system: SAFETY, messages: [{ role: "user", content }] }, effort ? { output_config: { effort: "low" } } : {}));
let last = null;
for (let attempt = 0; attempt < 3; attempt++) {
let r;
try {
r = await fetch("https://api.anthropic.com/v1/messages", {
method: "POST",
headers: { "content-type": "application/json", "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01" },
body: mk(),
});
} catch (e) { last = { error: "Claude: tarmoq xatosi " + String(e && e.message || e).slice(0, 120) }; await wait(800); continue; }
if (r.ok) {
const d = await r.json().catch(() => ({}));
if (d.stop_reason === "refusal") return { error: "Claude bu matnga javob bermadi (refusal)", code: "refused" };
const out = (d.content || []).filter((x) => x.type === "text").map((x) => x.text).join("");
if (out && !(d.stop_reason === "max_tokens" && attempt < 2 && limit < 32000)) return { text: out, model };
if (d.stop_reason === "max_tokens") limit = Math.min(limit * 2, 32000); const kinds = (d.content || []).map((x) => x.type).join(",") || "yoq"; last = { error: "Claude: böş javob (stop=" + (d.stop_reason || "?") + ", blok=" + kinds + ")" }; await wait(800); continue;
}
const t = await r.text().catch(() => "");
let m = t;
try { const j = JSON.parse(t); if (j.error) m = (j.error.type || "") + ": " + (j.error.message || ""); } catch (e) { /* plain text */ }
last = { error: "Claude " + r.status + " " + String(m).slice(0, 300), status: r.status };
if (r.status === 400 && effort && /output_config|effort/i.test(m)) { effort = false; continue; }
if (r.status === 400 && /max_tokens/i.test(m) && limit > 8192) { limit = 8192; continue; }
if (r.status === 429 || r.status === 529 || r.status >= 500) { await wait(1200 * (attempt + 1)); continue; }
break;
}
return last;
}

}

function newToken() {
const a = new Uint8Array(24);
crypto.getRandomValues(a);
return [...a].map((x) => x.toString(16).padStart(2, "0")).join("");
}
const KEY_ABC = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
function newKey() {
const a = new Uint8Array(12);
crypto.getRandomValues(a);
let s = "";
for (const x of a) s += KEY_ABC[x % 32];
return s.slice(0, 4) + "-" + s.slice(4, 8) + "-" + s.slice(8);
}
const normKey = (k) => String(k || "").toUpperCase().replace(/[^A-Z0-9]/g, "");

return k;
}

async function pushSend(env, v, eps) {
const key = await crypto.subtle.importKey("jwk", v.jwk, { name: "ECDSA", namedCurve: "P-256" }, false, ["sign"]);
const enc = (o) => b64u(new TextEncoder().encode(JSON.stringify(o)));
const jwt = {}, out = [];
for (const ep of eps) {
const aud = new URL(ep).origin;
if (!jwt[aud]) {
const un = enc({ typ: "JWT", alg: "ES256" }) + "." + enc({ aud, exp: Math.floor(Date.now() / 1000) + 12 * 3600, sub: "mailto:" + ownerEmail(env) });
jwt[aud] = un + "." + b64u(await crypto.subtle.sign({ name: "ECDSA", hash: "SHA-256" }, key, new TextEncoder().encode(un)));
}
let status = 0;
try {
const r = await fetch(ep, { method: "POST", body: "", headers: { TTL: "43200", Urgency: "high", Authorization: "vapid t=" + jwt[aud] + ", k=" + v.pub } });
status = r.status;
} catch (e) { status = 0; }
out.push({ ep, status });
}
return out;
}

if (ok.length) await env.DB.prepare(`UPDATE push SET sent = ?1, n = n + 1 WHERE endpoint IN (${q(ok, 2)})`).bind(now(), ...ok).run();
const bad = res.filter((r) => !(r.status >= 200 && r.status < 300) && r.status !== 404 && r.status !== 410);
if (bad.length) await env.DB.prepare("INSERT INTO logs (kind, msg, created) VALUES ('push', ?1, ?2)").bind("Bildirişnoma yuborilmadi: " + bad.length + " ta (javob " + bad[0].status + ")", now()).run();

}

async function resultsPut(b, env, send) {
const email = await authed(b, env);
if (!email) return send({ ok: false, code: "bad_token" }, 401);
const code = clean(b.code, 12), consent = b.consent ? 1 : 0;
const items = (Array.isArray(b.items) ? b.items : []).slice(0, 36).map((x) => ({
rid: clean(x.rid, 80), at: Math.max(0, Math.min(Date.now() + 86400000, Number(x.at) || 0)), track: x.track === "ielts" ? "ielts" : "cefr",
part: clean(x.part, 30), kind: clean(x.kind, 12), score: Number(x.score), max: Number(x.max) || 0,
crit: clean(typeof x.crit === "string" ? x.crit : JSON.stringify(x.crit || {}), 400), words: Math.max(0, parseInt(x.words, 10) || 0), ai: x.ai ? 1 : 0,
})).filter((x) => x.rid && x.at && isFinite(x.score));
for (let i = 0; i < items.length; i += 9) {
const chunk = items.slice(i, i + 9), args = [], marks = [];
chunk.forEach((x, k) => {
const o = k * 11;
marks.push("(" + Array.from({ length: 11 }, (_, j) => "?" + (o + j + 1)).join(",") + ")");
args.push(email, x.rid, x.at, x.track, x.part, x.kind, x.score, x.max, x.crit, x.words, x.ai);
});
await env.DB.prepare("INSERT OR IGNORE INTO results (email, rid, at, track, part, kind, score, max, crit, words, ai) VALUES " + marks.join(",")).bind(...args).run();
}
await env.DB.prepare("UPDATE users SET rcode = ?2, consent = ?3 WHERE email = ?1").bind(email, code, consent).run();
return send({ ok: true, saved: items.length });
}
async function adminResearch(req, env, send) {
if (!(await adminOk(req, env))) return send({ ok: false, code: "not_granted" }, 403);
const users = (await env.DB.prepare("SELECT email, nick, fullname, track, rcode, consent, created FROM users").all()).results || [];
const rows = (await env.DB.prepare("SELECT email, at, track, part, kind, score, max, crit, words, ai FROM results ORDER BY at LIMIT 30000").all()).results || [];
return send({ ok: true, users, rows });
}
